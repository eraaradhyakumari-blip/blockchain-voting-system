<script>

let blockchain=[];
let voted=[];

let votes={
Rahul:0,
Manish:0,
Ishan:0,
Aradhya:0
};

function simpleHash(text){
let hash=0;

for(let i=0;i<text.length;i++){
hash=((hash<<5)-hash)+text.charCodeAt(i);
hash=hash & hash;
}
return hash.toString();
}

function vote(){

let name=document.getElementById("voterName").value;
let age=document.getElementById("age").value;
let aadhar=document.getElementById("aadhar").value;
let candidate=document.getElementById("candidate").value;

if(name==""||age==""||aadhar==""||candidate==""){
alert("Fill all details");
return;
}

if(age<18){
alert("Only 18+ can vote");
return;
}

if(aadhar.length!=12){
alert("Aadhaar Number must be 12 digits");
return;
}

if(voted.includes(aadhar)){
alert("You have already voted");
return;
}

votes[candidate]++;

document.getElementById("rahul").innerHTML=votes.Rahul;
document.getElementById("manish").innerHTML=votes.Manish;
document.getElementById("ishan").innerHTML=votes.Ishan;
document.getElementById("aradhya").innerHTML=votes.Aradhya;

let previousHash=blockchain.length==0 ? "0" : blockchain[blockchain.length-1].hash;

let data=name+candidate+aadhar+Date.now();

let hash=simpleHash(data+previousHash);

let block={
index:blockchain.length+1,
name:name,
candidate:candidate,
previousHash:previousHash,
hash:hash
};

blockchain.push(block);

voted.push(aadhar);

displayChain();

document.getElementById("voterName").value="";
document.getElementById("age").value="";
document.getElementById("aadhar").value="";
document.getElementById("candidate").value="";
}

function displayChain(){

let output="";

for(let b of blockchain){

output+=`
<div class="block">
<b>Block:</b> ${b.index}<br>
<b>Voter:</b> ${b.name}<br>
<b>Candidate:</b> ${b.candidate}<br>
<b>Previous Hash:</b> ${b.previousHash}<br>
<b>Hash:</b> ${b.hash}
</div>
`;

}

document.getElementById("chain").innerHTML=output;

}
